# 元の hosts ファイルをバックアップ
Copy-Item -Path "C:\Windows\System32\drivers\etc\hosts" -Destination "C:\Windows\System32\drivers\etc\hosts.bak" -Force

# hosts ファイルをクリア
Set-Content -Path "C:\Windows\System32\drivers\etc\hosts" -Value ""

# 新しい hosts エントリを追加
$NewEntries = @(
    "10.10.50.134 test.local"
    "10.10.50.135 test.tech"
)

# ファイルに書き込む
Add-Content -Path "C:\Windows\System32\drivers\etc\hosts" -Value $NewEntries

Write-Output "hosts ファイルが更新されました！"
