import boto3
from botocore.exceptions import ClientError, EndpointConnectionError
from botocore.config import Config
import os

# AWS アカウントに対応するプロファイル名
profiles = ["dx_stg", "ai_role_play", "dx_prod", "common_id", "root"]

# チェック対象のリージョン
regions = ["ap-northeast-1", "us-east-1", "us-west-2"]

# 出力先フォルダ
output_dir = "lambda_export"
os.makedirs(output_dir, exist_ok=True)

# boto3 の通信設定（タイムアウト設定含む）
boto_config = Config(
    connect_timeout=5,
    read_timeout=10,
    retries={"max_attempts": 2}
)

def list_lambda_functions(profile, region):
    """
    指定されたプロファイルとリージョンのLambda関数を取得
    """
    try:
        session = boto3.Session(profile_name=profile, region_name=region)
        client = session.client("lambda", config=boto_config)

        functions = []
        paginator = client.get_paginator("list_functions")

        # ページネーション処理（最大10ページに制限）
        for i, page in enumerate(paginator.paginate()):
            if i >= 10:
                print(f"⚠️ ページが多すぎます：{profile} / {region}")
                break
            for fn in page.get("Functions", []):
                functions.append({
                    "FunctionName": fn.get("FunctionName"),
                    "Runtime": fn.get("Runtime", ""),
                    "LastModified": fn.get("LastModified", "")
                })

        return functions

    except EndpointConnectionError as e:
        print(f"❌ 接続失敗：{profile} / {region} → {str(e)}")
    except ClientError as e:
        print(f"❌ AWSエラー：{profile} / {region} → {str(e)}")
    except Exception as e:
        print(f"❌ 不明なエラー：{profile} / {region} → {str(e)}")

    return []  # 失敗時は空リストを返す

def save_to_file(profile, region, functions):
    """
    関数リストをファイルに保存（空ならスキップ）
    """
    if not functions:
        print(f"ℹ️ Lambda関数が見つかりませんでした：{profile} / {region}")
        return

    filename = f"{output_dir}/lambda_{profile}_{region}.txt"
    with open(filename, "w", encoding="utf-8") as f:
        f.write("FunctionName\tRuntime\tLastModified\n")
        for fn in functions:
            f.write(f"{fn['FunctionName']}\t{fn['Runtime']}\t{fn['LastModified']}\n")
    print(f"✅ 出力完了：{filename}")

def main():
    for profile in profiles:
        for region in regions:
            print(f"🔍 チェック中：プロファイル [{profile}], リージョン [{region}]")
            funcs = list_lambda_functions(profile, region)
            save_to_file(profile, region, funcs)
    print("✅ 全アカウント・リージョンの処理が完了しました。")

if __name__ == "__main__":
    main()
