#!/bin/bash

# ✅ 使用するAWSプロファイル（アカウント環境名に対応）
profiles=("dx_stg" "ai_role_play" "dx_prod" "common_id" "root")

# ✅ チェック対象のAWSリージョン（必要に応じて追加・削除OK）
regions=("ap-northeast-1" "us-east-1" "us-west-2")

# ✅ 出力先のフォルダを作成（存在しない場合）
mkdir -p lambda_export

# ✅ 各プロファイル・リージョンを順に処理
for profile in "${profiles[@]}"; do
  for region in "${regions[@]}"; do
    echo "📍 プロファイル [$profile] の リージョン [$region] を確認中..."

    # Lambda関数の一覧を取得し、指定ファイルに出力（タブ区切り）
    aws lambda list-functions \
      --profile $profile \
      --region $region \
      --query "Functions[*].[FunctionName,Runtime,LastModified]" \
      --output text >> lambda_export/lambda_${profile}_${region}.txt
  done
done

# ✅ 完了メッセージ
echo "✅ 全プロファイルのエクスポートが完了しました。出力ファイルは ./lambda_export/ に保存されています。"
