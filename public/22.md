import boto3
from botocore.exceptions import ClientError, EndpointConnectionError
from botocore.config import Config
import os
import csv

# プロファイル名リスト（アカウント環境）
profiles = ["dx_stg", "ai_role_play", "dx_prod", "common_id", "root"]

# チェック対象リージョン
regions = ["ap-northeast-1", "us-east-1", "us-west-2"]

# 出力先フォルダ
output_dir = "lambda_export_csv"
os.makedirs(output_dir, exist_ok=True)

# boto3 通信設定（タイムアウト付き）
boto_config = Config(
    connect_timeout=5,
    read_timeout=10,
    retries={"max_attempts": 2}
)

def list_lambda_functions(profile, region):
    """
    指定プロファイル・リージョン内の Lambda 関数一覧を取得
    """
    try:
        session = boto3.Session(profile_name=profile, region_name=region)
        client = session.client("lambda", config=boto_config)

        functions = []
        paginator = client.get_paginator("list_functions")

        for i, page in enumerate(paginator.paginate()):
            if i >= 10:
                print(f"⚠️ ページ数が多すぎます（{profile}/{region}）→ 最大10ページで中断")
                break
            for fn in page.get("Functions", []):
                functions.append({
                    "Profile": profile,
                    "Region": region,
                    "FunctionName": fn.get("FunctionName", ""),
                    "Runtime": fn.get("Runtime", ""),
                    "LastModified": fn.get("LastModified", "")
                })

        return functions

    except EndpointConnectionError as e:
        print(f"❌ 接続失敗：{profile}/{region} → {str(e)}")
    except ClientError as e:
        print(f"❌ AWSエラー：{profile}/{region} → {str(e)}")
    except Exception as e:
        print(f"❌ 不明なエラー：{profile}/{region} → {str(e)}")

    return []

def save_to_csv(profile, region, functions):
    """
    Lambda 関数一覧を CSV に保存
    """
    if not functions:
        print(f"ℹ️ Lambda関数なし：{profile}/{region}")
        return

    filename = f"{output_dir}/lambda_{profile}_{region}.csv"
    with open(filename, mode="w", encoding="utf-8", newline="") as f:
        writer = csv.DictWriter(f, fieldnames=["Profile", "Region", "FunctionName", "Runtime", "LastModified"])
        writer.writeheader()
        writer.writerows(functions)

    print(f"✅ CSV出力完了：{filename}")

def main():
    for profile in profiles:
        for region in regions:
            print(f"🔍 実行中：{profile} / {region}")
            funcs = list_lambda_functions(profile, region)
            save_to_csv(profile, region, funcs)
    print("✅ すべてのプロファイルとリージョンの処理が完了しました。")

if __name__ == "__main__":
    main()
