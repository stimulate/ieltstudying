import boto3
from botocore.exceptions import ClientError
import os

# アカウント環境名に対応したプロファイルリスト
profiles = ["dx_stg", "ai_role_play", "dx_prod", "common_id", "root"]

# チェック対象のAWSリージョン
regions = ["ap-northeast-1", "us-east-1", "us-west-2"]

# 出力フォルダ
output_dir = "lambda_export"
os.makedirs(output_dir, exist_ok=True)

def list_lambda_functions(profile, region):
    """
    指定したプロファイルとリージョンのLambda関数一覧を取得する関数
    """
    session = boto3.Session(profile_name=profile, region_name=region)
    client = session.client("lambda")
    functions = []

    paginator = client.get_paginator("list_functions")
    try:
        for page in paginator.paginate():
            for fn in page.get("Functions", []):
                functions.append({
                    "FunctionName": fn["FunctionName"],
                    "Runtime": fn.get("Runtime", ""),
                    "LastModified": fn.get("LastModified", "")
                })
    except ClientError as e:
        print(f"⚠️ エラー: プロファイル [{profile}] リージョン [{region}] の取得に失敗しました。{e}")
    
    return functions

def save_to_file(profile, region, functions):
    """
    関数リストをテキストファイルに保存する関数
    """
    filename = f"{output_dir}/lambda_{profile}_{region}.txt"
    with open(filename, "w", encoding="utf-8") as f:
        f.write("FunctionName\tRuntime\tLastModified\n")
        for fn in functions:
            f.write(f"{fn['FunctionName']}\t{fn['Runtime']}\t{fn['LastModified']}\n")
    print(f"✅ {filename} に保存しました。")

def main():
    for profile in profiles:
        for region in regions:
            print(f"📍 プロファイル [{profile}] のリージョン [{region}] を取得中...")
            funcs = list_lambda_functions(profile, region)
            save_to_file(profile, region, funcs)
    print("✅ 全てのエクスポートが完了しました。")

if __name__ == "__main__":
    main()
